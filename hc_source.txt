import os
import sys
import json
import subprocess
import importlib.util
from pathlib import Path

def check_agents_directory():
    """Verify .agents directory structure"""
    agents_dir = Path('.agents')
    if not agents_dir.exists():
        return False, ".agents directory not found"

    required_dirs = ['config', 'engine', 'rules', 'workflows', 'memory']
    missing = []
    for dir_name in required_dirs:
        if not (agents_dir / dir_name).exists():
            missing.append(dir_name)

    if missing:
        return False, f"Missing directories: {', '.join(missing)}"

    return True, "Directory structure OK"

def check_core_dependencies():
    """Check if core Python dependencies are available"""
    required_modules = ['yaml', 'dotenv', 'jsonschema']
    missing = []

    for module in required_modules:
        if importlib.util.find_spec(module) is None:
            missing.append(module)

    if missing:
        return False, f"Missing modules: {', '.join(missing)}"

    return True, "Core dependencies OK"

def check_agent_configs():
    """Validate agent configuration files"""
    config_dir = Path('.agents/config')
    if not config_dir.exists():
        return False, "Config directory not found"

    required_configs = ['brain.md', 'sentinel.md', 'boom.md']
    missing = []

    for config in required_configs:
        if not (config_dir / config).exists():
            missing.append(config)

    if missing:
        return False, f"Missing configs: {', '.join(missing)}"

    return True, "Agent configs OK"

def check_engine_integrity():
    """Test if the agent engine can start"""
    engine_dir = Path('.agents/engine')
    main_py = engine_dir / 'main.py'

    if not main_py.exists():
        return False, "Engine main.py not found"

    try:
        # Try to import the main module
        sys.path.insert(0, str(engine_dir))
        spec = importlib.util.spec_from_file_location("main", main_py)
        if spec is None or spec.loader is None:
            return False, "Cannot load engine module"

        # Basic syntax check
        compile(spec.loader.get_source(), str(main_py), 'exec')
        return True, "Engine integrity OK"

    except Exception as e:
        return False, f"Engine check failed: {e}"
    finally:
        if str(engine_dir) in sys.path:
            sys.path.remove(str(engine_dir))

def run_health_check():
    """Run comprehensive health check"""
    print("🩺 Jules Code Team - Health Check")
    print("=" * 40)

    checks = [
        ("Directory Structure", check_agents_directory),
        ("Core Dependencies", check_core_dependencies),
        ("Agent Configurations", check_agent_configs),
        ("Engine Integrity", check_engine_integrity),
    ]

    all_passed = True
    for name, check_func in checks:
        print(f"Checking {name}...", end=" ")
        try:
            passed, message = check_func()
            if passed:
                print("✅")
            else:
                print("❌")
                print(f"   {message}")
                all_passed = False
        except Exception as e:
            print("❌")
            print(f"   Error: {e}")
            all_passed = False

    print()
    if all_passed:
        print("🎉 All checks passed! System is healthy.")
        return 0
    else:
        print("⚠️  Some checks failed. Run 'squad --diagnose' for detailed help.")
        return 1

if __name__ == "__main__":
    sys.exit(run_health_check())
